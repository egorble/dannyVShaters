// Hàm trợ giúp để gửi một mutation GraphQL
async function sendGraphQLMutation(url, query, variables = {}) {
    try {
        const response = await fetch(url, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify({
                query,
                variables,
            }),
        });
        
        // Lấy raw text response trước khi parse JSON
        const responseText = await response.text();
        console.log("Raw GraphQL Mutation Response:", responseText); // Log raw response

        // Cố gắng parse JSON từ raw text
        const data = JSON.parse(responseText);

        if (data.errors) {
            console.error("GraphQL Errors:", data.errors);
            throw new Error(`GraphQL Errors: ${JSON.stringify(data.errors)}`);
        }
        return data;
    } catch (error) {
        console.error("Lỗi khi gửi GraphQL Mutation:", error);
        throw error;
    }
}

// Endpoint để tạo Chain ID mới (từ Faucet)
app.post('/create-chain', async (req, res) => {
    console.log('Yêu cầu tạo Chain ID mới đã nhận.');

    let newMatchChainId;
    try {
        console.log('Đang yêu cầu Chain ID mới từ Faucet bằng lệnh curl...');

        const MAX_RETRIES = 5;
        let retries = 0;
        let faucetData = null;

        // Địa chỉ owner placeholder. Trong thực tế, đây sẽ là địa chỉ công khai của một ví Linera.
        // Bạn có thể thay thế bằng địa chỉ ví Linera thực của mình nếu có.
        const ownerAddress = "0x0000000000000000000000000000000000000000"; // Đảm bảo đây là địa chỉ ví Linera hợp lệ

        while (retries < MAX_RETRIES && !faucetData) {
            try {
                await delay(500); // Chờ 0.5 giây trước mỗi lần thử

                const curlCommand = `curl -X POST ${FAUCET_URL} -H "Content-Type: application/json" -d '{"query":"mutation { claim(owner: \\"${ownerAddress}\\") { chainId } }"}'`;

                const { stdout: curlOutput, stderr: curlError } = await execPromise(curlCommand);

                console.log(`Curl stdout: [${curlOutput.trim()}]`);
                if (curlError) {
                    console.error(`Curl stderr (error here): ${curlError}`);
                }

                if (!curlOutput.trim()) {
                    throw new Error("Empty or invalid response from Faucet via curl.");
                }

                faucetData = JSON.parse(curlOutput);

                if (faucetData.errors) {
                    console.error(`Lỗi từ Faucet (qua curl): ${JSON.stringify(faucetData.errors)}`);
                    throw new Error(`Faucet returned error: ${JSON.stringify(faucetData.errors)}`);
                }

                newMatchChainId = faucetData.data.claim.chainId;
                console.log(`Đã nhận Chain ID mới từ Faucet: ${newMatchChainId}`);
                break;

            } catch (error) {
                retries++;
                console.warn(`Lỗi khi gọi Faucet (lần thử ${retries}/${MAX_RETRIES}): ${error.message}`);
                if (retries >= MAX_RETRIES) {
                    console.error(`Không thể lấy Chain ID từ Faucet sau ${MAX_RETRIES} lần thử.`);
                    return res.status(500).json({ error: 'Failed to get Chain ID from Faucet', details: error.message });
                }
                await delay(2000 * retries);
            }
        }

        if (!newMatchChainId) {
            throw new Error("Không thể lấy Chain ID mới từ Faucet.");
        }

        // Tạm thời bỏ qua bước khởi tạo ứng dụng XFighter trên Chain ID mới
        // Nếu bạn muốn khởi tạo XFighter trên mỗi chain mới, hãy bỏ comment và cấu hình đúng.
        /*

                                   `--json-argument 'null' ` + // Tham số khởi tạo cho xfighter, có thể thay đổi
                                   `--json-parameters '{ "leaderboard_chain_id": "${LEADERBOARD_CHAIN_ID}", "leaderboard_application_id": "${LEADERBOARD_APP_ID}" }' ` +
                                   `--json-chain-id ${newMatchChainId}`;

    } catch (error) {
        console.error('Lỗi trong quá trình tạo Chain ID:', error);
        res.status(500).json({ error: 'Internal server error', details: error.message });

// --- API để mô phỏng XFighter gửi điểm (Record Match Result) ---
app.post('/xfighter/record-match', async (req, res) => {
    const {
        matchId, player1Username, player2Username, winnerUsername, loserUsername,
        durationSeconds, timestamp, player1Score, player2Score, mapName, matchType
    } = req.body;

    // Sử dụng biến XFIGHTER_SERVICE_URL đã được cập nhật
    const mutation = `
        mutation RecordMatch($result: MatchResultInput!) {
    recordMatchResult(result: $result)
    }`;
    const variables = {
        result: { // Đã thay đổi từ input: sang result:
            matchId, player1Username, player2Username, winnerUsername, loserUsername,
            durationSeconds, timestamp, player1Score, player2Score, mapName, matchType
        }
    };

    try {
        // Gọi sendGraphQLMutation với URL đã được cập nhật
        const result = await sendGraphQLMutation(XFIGHTER_SERVICE_URL, mutation, variables);
        console.log("XFighter Match Recorded Result:", result);

        // Chờ một chút để Linera xử lý tin nhắn xuyên chuỗi đến Leaderboard
        await new Promise(resolve => setTimeout(resolve, 3000));

        res.json({ success: true, message: 'Trận đấu XFighter đã được ghi lại.', result });
    } catch (error) {
        console.error('Lỗi khi ghi lại trận đấu XFighter:', error);
        res.status(500).json({ success: false, message: 'Không thể ghi lại trận đấu.', error: error.message });
    }
});

// --- API để truy vấn Leaderboard ---
app.get('/leaderboard/get-state', async (req, res) => {
    // Sửa đổi câu truy vấn GraphQL để bao gồm totalMatches
    const query = `
        query GetLeaderboard {
            leaderboard {
                userId
                score
                totalMatches
            }
        }
    `;
    try {
        // Gọi sendGraphQLQuery với URL đã được cập nhật
        const result = await sendGraphQLQuery(LEADERBOARD_SERVICE_URL, query);
        // Cập nhật logic xử lý phản hồi để hiển thị thông tin mới
        console.log("Current Leaderboard:");
        result.data.leaderboard.forEach(entry => {
            console.log(`- User: ${entry.userId}, Score: ${entry.score}, Matches: ${entry.totalMatches}`);
        });

        res.json({ success: true, leaderboard: result.data.leaderboard });
    } catch (error) {
        console.error('Lỗi khi truy vấn Leaderboard:', error);
        res.status(500).json({ success: false, message: 'Không thể lấy Leaderboard.', error: error.message });
    }
});
